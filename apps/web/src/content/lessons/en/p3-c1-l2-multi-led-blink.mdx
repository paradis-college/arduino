<LessonChapter title="Multiple Outputs" id="multiple-outputs">

### The Hard Way

```cpp
void setup() {
  pinMode(9, OUTPUT);
  pinMode(10, OUTPUT);
  pinMode(11, OUTPUT);
}
```

### The Better Way - Arrays

```cpp
int ledPins[] = {9, 10, 11};
int numLeds = 3;

void setup() {
  for (int i = 0; i < numLeds; i++) {
    pinMode(ledPins[i], OUTPUT);
  }
}
```

<InfoBox type="tip">
  Arrays make it easy to add more LEDs - just update the array and count!
</InfoBox>

<LessonChapter title="All Blink Pattern" id="all-blink">

### Blink All LEDs Together

```cpp
void allBlink() {
  // All on
  for (int i = 0; i < numLeds; i++) {
    digitalWrite(ledPins[i], HIGH);
  }
  delay(500);
  
  // All off
  for (int i = 0; i < numLeds; i++) {
    digitalWrite(ledPins[i], LOW);
  }
  delay(500);
}
```

</LessonChapter>

<LessonChapter title="Chase Pattern" id="chase-pattern">

### One LED at a Time

```cpp
void chasePattern() {
  for (int i = 0; i < numLeds; i++) {
    // Turn current LED on
    digitalWrite(ledPins[i], HIGH);
    delay(200);
    // Turn current LED off
    digitalWrite(ledPins[i], LOW);
  }
}
```

### Bounce/Ping-Pong Pattern

```cpp
void pingPongPattern() {
  // Forward
  for (int i = 0; i < numLeds; i++) {
    digitalWrite(ledPins[i], HIGH);
    delay(150);
    digitalWrite(ledPins[i], LOW);
  }
  // Backward
  for (int i = numLeds - 2; i > 0; i--) {
    digitalWrite(ledPins[i], HIGH);
    delay(150);
    digitalWrite(ledPins[i], LOW);
  }
}
```

<LessonChapter title="Alternate Pattern" id="alternate-pattern">

### Every Other LED

```cpp
void alternatePattern() {
  // Odd LEDs on
  for (int i = 0; i < numLeds; i += 2) {
    digitalWrite(ledPins[i], HIGH);
  }
  for (int i = 1; i < numLeds; i += 2) {
    digitalWrite(ledPins[i], LOW);
  }
  delay(300);
  
  // Even LEDs on
  for (int i = 0; i < numLeds; i += 2) {
    digitalWrite(ledPins[i], LOW);
  }
  for (int i = 1; i < numLeds; i += 2) {
    digitalWrite(ledPins[i], HIGH);
  }
  delay(300);
}
```
