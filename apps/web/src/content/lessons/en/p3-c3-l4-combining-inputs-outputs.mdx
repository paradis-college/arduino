### If-Then Rules

```cpp
void loop() {
  bool btn1 = digitalRead(BTN_1) == LOW;
  bool btn2 = digitalRead(BTN_2) == LOW;
  int pot = analogRead(POT_PIN);
  
  // Rule 1: Button 1 turns on LED 1
  digitalWrite(LED_1, btn1 ? HIGH : LOW);
  
  // Rule 2: Button 2 turns on LED 2
  digitalWrite(LED_2, btn2 ? HIGH : LOW);
  
  // Rule 3: Both buttons turn on LED 3
  digitalWrite(LED_3, (btn1 && btn2) ? HIGH : LOW);
}
```

### Potentiometer as Intensity Control

```cpp
void loop() {
  bool powerOn = digitalRead(BTN_1) == LOW;
  int brightness = map(analogRead(POT_PIN), 0, 1023, 0, 255);
  
  if (powerOn) {
    analogWrite(LED_1, brightness);  // Pot controls brightness when on
  } else {
    analogWrite(LED_1, 0);           // Off when button not pressed
  }
}
```

### Potentiometer as Mode Selector

```cpp
void loop() {
  int pot = analogRead(POT_PIN);
  int mode = pot / 342;  // Divide 1023 into 3 zones (0, 1, 2)
  
  // Turn off all LEDs first
  digitalWrite(LED_1, LOW);
  digitalWrite(LED_2, LOW);
  digitalWrite(LED_3, LOW);
  
  // Light LED based on mode
  switch (mode) {
    case 0:
      digitalWrite(LED_1, HIGH);
      break;
    case 1:
      digitalWrite(LED_2, HIGH);
      break;
    case 2:
      digitalWrite(LED_3, HIGH);
      break;
  }
}
```

### Button as Mode Cycle

```cpp
int currentMode = 0;
const int NUM_MODES = 3;

void handleModeButton() {
  currentMode = (currentMode + 1) % NUM_MODES;
}
```

### Simple Lighting Control Panel

```cpp
const int BTN_POWER = 2;
const int BTN_MODE = 3;
const int POT_BRIGHTNESS = A0;
const int LEDS[] = {9, 10, 11};

bool powerOn = false;
int mode = 0;

void loop() {
  // Read inputs
  if (buttonJustPressed(BTN_POWER)) {
    powerOn = !powerOn;
  }
  if (buttonJustPressed(BTN_MODE)) {
    mode = (mode + 1) % 3;
  }
  int brightness = map(analogRead(POT_BRIGHTNESS), 0, 1023, 0, 255);
  
  // Apply outputs
  if (!powerOn) {
    allLedsOff();
  } else {
    applyMode(mode, brightness);
  }
}
```

<InfoBox type="tip">
  Build up complexity gradually. Start with one input and one output, then add more once that works!
